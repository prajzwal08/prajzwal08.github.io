% The same blue as the website and the .docx CV.
\definecolor{cvblue}{RGB}{0,112,192}

% hidelinks, so hyperref colours nothing on its own and every link inherits
% the colour of the text around it. Paper titles are therefore black: with
% nine of them, a blue title each was most of the colour on the page.
\hypersetup{hidelinks}

% \link is the header only: the website and the email addresses. It is the
% only blue on the page.
\newcommand{\link}[2]{\href{#1}{\textcolor{cvblue}{#2}}}

% Everything in the body. Underlined rather than coloured, so a reader can see
% there is something to click without the page turning blue. \uline comes from
% ulem rather than \underline because a paper title is long and has to be able
% to break across lines.
\newcommand{\ulink}[2]{\href{#1}{\uline{#2}}}

\pagestyle{empty}
\setlength{\parindent}{0pt}
\setlength{\parskip}{0pt}

% ---------------------------------------------------------------------------
% Section: a small-caps label in the left margin, content to the right.
% ---------------------------------------------------------------------------
\newlength{\cvlabelwidth}
\setlength{\cvlabelwidth}{0.95in}

\newenvironment{cvsection}[1]{%
  \begin{list}{}{%
    \setlength{\leftmargin}{\cvlabelwidth}%
    \addtolength{\leftmargin}{0.12in}%
    \setlength{\labelwidth}{\cvlabelwidth}%
    \setlength{\labelsep}{0.12in}%
    \setlength{\itemindent}{0pt}%
    \setlength{\listparindent}{0pt}%
    \setlength{\topsep}{0.02in}%
    \setlength{\parsep}{0pt}%
    \setlength{\itemsep}{0pt}%
    % No \smash here. Smashing the label to zero height stops a two-line label
    % from stretching the first line of the content beside it — but it also
    % lets that second label line print on top of the next section's label,
    % which is exactly what "Research Interests" did to "Education". Better to
    % keep every label to one line: see \cvlabelwidth above.
    % Blue, matching the website's section headings. The column is wide enough
    % for the longest label, "Scholarships", to stay on one line — a label that
    % wraps stretches the first line of the content beside it.
    \renewcommand{\makelabel}[1]{%
      \parbox[t]{\cvlabelwidth}{\raggedright\textcolor{cvblue}{\textsc{##1}}}}%
  }%
  \item[#1]%
}{\end{list}}

% One entry: bold title, italic institution, italic dates flushed right.
% Facing minipages rather than \hfill, so a long title wraps in its own
% column instead of shoving the date onto the next line.
\newcommand{\entry}[3]{%
  \noindent
  \begin{minipage}[t]{0.72\linewidth}\raggedright\textbf{#1}\textit{#2}\end{minipage}%
  \hfill
  \begin{minipage}[t]{0.26\linewidth}\raggedleft\textit{#3}\end{minipage}%
  \par
}

% One achievement under an entry, marked with a dash so that two of them read
% as two rather than as one wrapped sentence. The hanging indent keeps a
% wrapped line clear of the dash.
\newcommand{\detail}[1]{%
  \hspace*{0.55em}\textendash\hspace{0.4em}%
  \parbox[t]{\dimexpr\linewidth-1.75em}{\raggedright #1}\par}

% One publication. The title carries the link, so no "doi" is printed.
\newcommand{\pub}[1]{\vspace{0.015in}#1\par}

% A heading inside a section, e.g. "Peer-reviewed".
\newcommand{\subhead}[1]{\vspace{0.035in}\textit{#1}\par}

\newcommand{\gap}{\vspace{0.02in}}
